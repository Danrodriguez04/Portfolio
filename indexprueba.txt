    function drawProjects(projects) {
        const projectContainer = document.querySelector('#project-list');
        const skillsContainer = document.querySelector('#skills-list');
        const skills = new Set();

        console.log(projects);
        for (const project of projects) {
            createCardProject(project, projectContainer);
            skills.add(...project.getSkills());
        }

        for (const skill of skills) {
            createSkillSection(skill, skillsContainer);
        }

    }

    function createSkillSection(skill, skillContainer) {
        const containerSkill = document.createElement('DIV');
        const iconSkill = document.createElement('LI')
        const iconName = document.createElement('P');

        iconSkill.classList.add(...skill.getClassIcon().split(' '), 'iconSkill');
        iconName.innerHTML = skill.getName();

        containerSkill.append(iconSkill, iconName);

        skillContainer.append(containerSkill);
    }

    function createCardProject(project, projectContainer) {
        const card = document.createElement('DIV');
        const content = document.createElement('DIV');
        const title = document.createElement('h3');
        const description = document.createElement('p');
        const button = document.createElement('A');

        card.className = 'card';
        card.style.backgroundImage = `url(${project.getImageLocation()})`;
        title.innerHTML = project.getName();
        description.innerHTML = project.getDescription();
        button.innerHTML = 'Ver proyecto !';
        button.href = '#'
        button.className = 'see-more';

        content.append(title);
        content.append(description);
        content.append(button);
        card.append(content);

        projectContainer.append(card);
    }
